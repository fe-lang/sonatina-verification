This module is meant to be used for typechecking of the Prim namespace,
the more compact export module is Prim.Prelude

```agda
{-# OPTIONS --safe --erased-cubical #-}

module Prim.Everything where

open import Prim.Type public
open import Prim.Data public
open import Prim.Interval.Core public
open import Prim.Equiv public
open import Prim.Fiber public
open import Prim.Interval public
open import Prim.Literals public
open import Prim.Pi public
open import Prim.Reasoning public
open import Prim.SSet public


```
